\subsection{Degradation Model (Wald Protocol)}
HR = 4-band 10~m Sentinel-2 tile (B2/B3/B4/B8). LR = \texttt{ALSAT2BDegradation.forward\_ms\_only}:
\begin{enumerate}
    \item Gaussian MTF blur ($\sigma = 0.35$ at 10~m)
    \item Area $\times$4 downsample (10~m $\to$ 40~m)
    \item Poisson-Gaussian noise (SNR 20~dB)
    \item 10-bit quantization
\end{enumerate}
Config: \texttt{configs/degradation/alsat2b.yaml}. Deterministic with seed 42.

\subsection{Evaluation Protocol}
\begin{itemize}
    \item \textbf{Fidelity:} PSNR, SSIM, MSE, MAE, ERGAS, SAM, SCI, wavelet HH energy, GLCM entropy (via \texttt{src/evaluation/metrics.py::evaluate\_sr})
    \item \textbf{Hallucination:} per-category detection rates + confidence (via \texttt{detect\_all})
    \item \textbf{Baseline:} bicubic $\times$4 recompute on the same pairs for fairness
\end{itemize}
